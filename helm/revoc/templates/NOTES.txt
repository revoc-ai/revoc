Emergency kill-switch (internal port {{ .Values.proxy.ports.internal }} — never exposed via Ingress)
──────────────────────────────────────────────────────────────────────────────
  kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "revoc.proxy.fullname" . }}-internal {{ .Values.proxy.ports.internal }}:{{ .Values.proxy.ports.internal }}

  # Activate (stops ALL traffic across every instance within ~50ms):
  curl -s -X POST http://localhost:{{ .Values.proxy.ports.internal }}/internal/emergency/kill-switch/activate \
    -H "X-Admin-Key: <REVOC_AUTH__ADMIN_KEY>" -H "Content-Type: application/json" \
    -d '{"scope":"global","reason":"incident-response","activated_by":"oncall"}'

  # Status:
  curl -s http://localhost:{{ .Values.proxy.ports.internal }}/internal/emergency/kill-switch/status \
    -H "X-Admin-Key: <REVOC_AUTH__ADMIN_KEY>"

  # Deactivate:
  curl -s -X POST http://localhost:{{ .Values.proxy.ports.internal }}/internal/emergency/kill-switch/deactivate \
    -H "X-Admin-Key: <REVOC_AUTH__ADMIN_KEY>" -H "Content-Type: application/json" \
    -d '{"scope":"global"}'

  Or via the CLI from a proxy pod:
    kubectl exec -n {{ .Release.Namespace }} deploy/{{ include "revoc.proxy.fullname" . }} -- \
      revoc-proxy emergency activate --scope global --reason "incident-response" --admin-key <ADMIN_KEY>
